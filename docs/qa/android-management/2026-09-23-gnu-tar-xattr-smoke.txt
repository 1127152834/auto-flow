import asyncio
import io
import json
import tarfile
from uuid import uuid4

from autoflow.providers.android.mac_runtime import run


async def main():
    root = '/tmp/autoflow-gnu-tar-xattr-' + uuid4().hex
    setup = "import os,sys; r=sys.argv[1]; os.makedirs(r+'/source/_data'); os.makedirs(r+'/target'); p=r+'/source/_data/proof'; open(p,'wb').write(b'proof'); os.setxattr(p,'user.autoflow_probe',b'metadata'); os.chmod(p,0o640)"
    await run(['limactl','shell','--workdir=/tmp','autoflow-redroid','sudo','python3','-c',setup,root])
    try:
        data = await run(['limactl','shell','--workdir=/tmp','autoflow-redroid','sudo','tar','--format=posix','--xattrs','--xattrs-include=*','--acls','--selinux','--numeric-owner','--transform=s@^_data@data@','-C',root+'/source','-cf','-','_data'],30)
        with tarfile.open(fileobj=io.BytesIO(data)) as archive:
            members = [{'name':item.name,'pax':item.pax_headers,'uid':item.uid,'gid':item.gid,'mode':oct(item.mode)} for item in archive]
        await run(['limactl','shell','--workdir=/tmp','autoflow-redroid','sudo','tar','--xattrs','--xattrs-include=*','--acls','--selinux','--numeric-owner','--same-owner','--same-permissions','--strip-components=1','-C',root+'/target','-xf','-'],30,data)
        check = "import json,os,sys; p=sys.argv[1]+'/target/proof'; print(json.dumps({'content':open(p,'rb').read().decode(),'xattr':os.getxattr(p,'user.autoflow_probe').decode(),'mode':oct(os.stat(p).st_mode&0o777)}))"
        result = json.loads(await run(['limactl','shell','--workdir=/tmp','autoflow-redroid','sudo','python3','-c',check,root]))
        print(json.dumps({'members':members,'restored':result,'status':'passed'},indent=2))
    finally:
        await run(['limactl','shell','--workdir=/tmp','autoflow-redroid','sudo','rm','-rf',root])


asyncio.run(main())
