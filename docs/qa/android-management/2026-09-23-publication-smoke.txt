import asyncio
import io
import json
import shlex
import tarfile
from pathlib import Path
from uuid import uuid4
from autoflow.application.android.backups import AndroidBackupService
from autoflow.application.android.management import AndroidManagement
from autoflow.bootstrap.android_prepare import prepare
from autoflow.infrastructure.database.android import SqlAlchemyDeviceRepository
from autoflow.infrastructure.database.android_resources import AndroidResourceRepository
from autoflow.infrastructure.database.android_operations import SqlAlchemyAndroidOperationRepository
from autoflow.infrastructure.database.session import create_session_factory
from autoflow.infrastructure.filesystem.android_paths import android_runtime_root
from autoflow.infrastructure.filesystem.paths import AppPaths
from autoflow.providers.android.mac_runtime import MacAndroidRuntime, docker
from autoflow.providers.android.management import verify

async def operation(runtime, repo, device, action, delete=False):
    service = AndroidManagement(repo, runtime)
    service.operate(device['deviceId'], {'requestId': str(uuid4()), 'action': action, 'deleteData': delete})
    await service.task
    result = repo.get(device['deviceId'])
    assert result['operation']['state'] == 'succeeded', result
    return result

def metadata(data):
    with tarfile.open(fileobj=io.BytesIO(data), mode='r:') as archive:
        return {m.name: {'uid': m.uid, 'gid': m.gid, 'mode': oct(m.mode), 'type': m.type.decode(), 'link': m.linkname, 'content': archive.extractfile(m).read().decode() if m.isreg() else None} for m in archive.getmembers() if '/autoflow-metadata/' in m.name}

async def main():
    data_dir = Path('/tmp/autoflow-publication-20260923')
    if data_dir.exists():
        raise RuntimeError('Refusing to reuse workspace')
    if (await docker('ps', '-q')).strip():
        raise RuntimeError('Other containers running; leave untouched')
    source = await prepare(data_dir, Path('/Users/zhangtiancheng/.autoflow/android-runtime/scrcpy-macos-aarch64-v3.3.4.tar.gz'), False)
    paths = AppPaths.from_data_dir(data_dir)
    sessions = create_session_factory(paths.database)
    repo = SqlAlchemyDeviceRepository(sessions)
    runtime = MacAndroidRuntime(android_runtime_root(), paths.workspace)
    backups = AndroidBackupService(AndroidResourceRepository(sessions), paths.workspace, SqlAlchemyAndroidOperationRepository(sessions))
    target = None
    record = None
    report = {'sourceDeviceId': source['deviceId'], 'workspaceId': runtime.workspace_id, 'imageId': source['imageId'], 'cleanup': []}
    try:
        runtime.lock()
        try:
            await runtime.connect(source, lambda: repo.save(source))
            await docker('exec', '-u', '0', source['containerId'], 'sh', '-c', 'mkdir /data/local/tmp/autoflow-metadata && printf metadata-proof > /data/local/tmp/autoflow-metadata/file && chown 10001:2000 /data/local/tmp/autoflow-metadata/file && chmod 640 /data/local/tmp/autoflow-metadata/file && ln /data/local/tmp/autoflow-metadata/file /data/local/tmp/autoflow-metadata/hard && ln -s file /data/local/tmp/autoflow-metadata/relative && ln -s /data/local/tmp/autoflow-metadata/file /data/local/tmp/autoflow-metadata/absolute')
        finally:
            await runtime.disconnect()
            runtime.unlock()
        source = await operation(runtime, repo, source, 'stop')
        record = await backups.create_with_runtime(source, None, runtime, 'metadata-backup', source['generation'])
        before = metadata((Path(record['path']) / 'data.tar').read_bytes())
        report.update(backupId=record['id'], backupBytes=record['bytes'], sourceMetadata=before)
        service = AndroidManagement(repo, runtime)
        target = service.create({'deviceId': str(uuid4()), 'name': 'metadata-restore', 'imageId': source['imageId'], 'width': 720, 'height': 1280, 'dpi': 320, 'cpu': 1, 'memoryMb': 1536, 'start': False})
        await service.task
        target = repo.get(target['deviceId'])
        assert target['operation']['state'] == 'succeeded', target
        report['targetDeviceId'] = target['deviceId']
        await backups.restore_data(record['id'], target, runtime)
        restored = metadata(await runtime.backup_volume(target))
        report['restoredMetadata'] = restored
        assert restored == before, (before, restored)
        assert before['data/local/tmp/autoflow-metadata/file']['uid'] == 10001
        assert before['data/local/tmp/autoflow-metadata/file']['gid'] == 2000
        assert before['data/local/tmp/autoflow-metadata/file']['mode'] == '0o640'
        target = await operation(runtime, repo, target, 'start')
        runtime.lock()
        try:
            await runtime.connect(target, lambda: repo.save(target))
            readback = (await runtime._adb('shell', 'sh', '-c', shlex.quote('cat /data/local/tmp/autoflow-metadata/relative; cat /data/local/tmp/autoflow-metadata/absolute; cat /data/local/tmp/autoflow-metadata/hard'))).decode().strip()
            assert readback == 'metadata-proof' * 3, readback
            report['bootedReadback'] = readback
        finally:
            await runtime.disconnect()
            runtime.unlock()
        after_source = metadata(await runtime.backup_volume(source))
        assert after_source == before
        report.update(sourceUnchanged=True, status='passed')
    except BaseException as error:
        report.update(status='failed', error=repr(error))
        raise
    finally:
        for device in (source, target):
            if device is None:
                continue
            current = repo.get(device['deviceId'])
            if current.get('control') == 'recovery_required':
                current = await operation(runtime, repo, current, 'recover')
            current = await operation(runtime, repo, current, 'delete', True)
            containers, volumes = await verify(current, runtime.workspace_id)
            report['cleanup'].append({'deviceId': current['deviceId'], 'containers': len(containers), 'volumes': len(volumes), 'deleted': current.get('deleted')})
            assert current.get('deleted') and not containers and not volumes
        if record:
            backups.delete(record['id'])
        sessions.dispose()
        print(json.dumps(report, ensure_ascii=False, indent=2))

asyncio.run(main())
